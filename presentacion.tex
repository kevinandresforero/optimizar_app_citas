\begin{frame}{Modelo Lotka-Volterra para App de Citas}
  \centering
  \includegraphics[width=0.75\textwidth]{modelo_ecs.png}

  \vspace{0.3cm}
  \small
  El sistema modela la interacción entre perfiles disponibles ($x$) y usuarios
  activos ($y$). El objetivo es encontrar parámetros que maximicen la rentabilidad
  manteniendo un equilibrio estable.
\end{frame}

\begin{frame}{Acrónimos — Optimizadores}
  \centering
  \begin{itemize}
    \item[\textbf{DE}] \textbf{D}ifferential \textbf{E}volution
      \newline \hspace{1.2em} Búsqueda evolutiva global con población de individuos
    \bigskip
    \item[\textbf{SGD}] \textbf{S}tochastic \textbf{G}radient \textbf{D}escent
      \newline \hspace{1.2em} Gradiente descendente estocástico con momentum
    \bigskip
    \item[\textbf{ANFIS}] \textbf{A}daptive \textbf{N}euro-\textbf{F}uzzy
      \newline \hspace{1.2em} \textbf{I}nference \textbf{S}ystem

  \end{itemize}
\end{frame}

\begin{frame}{Notación — Métricas de evaluación}
  \centering
  \begin{itemize}
    \item[\textbf{CV}] \textbf{C}oefficient of \textbf{V}ariation
      \newline \hspace{1.2em} $\displaystyle CV = \frac{\sigma(y)}{\mu(y)}$ en estado estable
      \newline \hspace{1.2em} Mide la estabilidad del sistema ($<0.3$ = estable)
    \bigskip
    \item[\boldmath{$x^{*}$}] Equilibrio de perfiles
      \newline \hspace{1.2em} $\displaystyle x^{*} = \frac{d}{c}$
    \bigskip
    \item[\boldmath{$y^{*}$}] Equilibrio de usuarios
      \newline \hspace{1.2em} $\displaystyle y^{*} = \frac{a}{b}$
  \end{itemize}

\end{frame}

\begin{frame}{DE — ¿Cómo funciona?}

  \vspace{0.3cm}

    \begin{block}{Analogía}

    \end{block}
    \column{0.5\textwidth}
    \begin{block}{Después del DE}

\begin{frame}{L-BFGS-B — ¿Cómo funciona?}
  \centering
  \textbf{Optimizador local basado en gradiente — el \"refinador\"}

  \vspace{0.2cm}
  \begin{tikzpicture}[node distance=0.3cm]
    \node[draw, fill=ForestGreen!20, rounded corners, inner sep=6pt, align=center] (start) {\textbf{Punto de partida}\\[0.05cm] \tiny La mejor solución del DE};
    \node[draw, fill=ForestGreen!40, rounded corners, inner sep=6pt, align=center, right=1.5cm of start] (grad) {\textbf{Calcula pendiente}\\[0.05cm] \tiny Dirección cuesta abajo};
    \node[draw, fill=ForestGreen!60, rounded corners, inner sep=6pt, align=center, right=1.5cm of grad] (step) {\textbf{Avanza un paso}\\[0.05cm] \tiny Sin salirse de los límites};
    \draw[->, thick] (start.east) -- (grad.west);
    \draw[->, thick] (grad.east) -- (step.west);
    \draw[->, thick, rounded corners] (step.north) -- ++(0,1) -- ++(-6.5,0) -- (start.north);
    \node[above=0.7cm of grad, scale=0.6] {\bfseries Repetir hasta converger};
  \end{tikzpicture}

  \vspace{0.25cm}
  \begin{columns}[t]
    \column{0.5\textwidth}
    \begin{block}{¿Qué significa el nombre?}
      \scriptsize
      \textbf{L}imited memory: solo recuerda los últimos pasos \\[0.05cm]
      \textbf{BFGS}: método que aproxima la curvatura \\[0.05cm]
      \textbf{B}ox: respeta límites (caja) en cada parámetro
    \end{block}
    \column{0.5\textwidth}
    \begin{block}{Analogía}
      \scriptsize
      Como bajar una montaña con los ojos cerrados:
      sientes la pendiente con el pie (\textit{gradiente}),
      das un paso en esa dirección, y repites hasta
      llegar al valle. Las \textit{cajas} son cercos
      que impiden salir del camino permitido.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[shrink]{L-BFGS-B vs SGD — Diferencias clave}
  \centering
  \textbf{Dos formas de bajar la montaña}

  \vspace{0.2cm}
  \begin{columns}[t]
    \column{0.5\textwidth}
    \begin{block}{\textcolor{ForestGreen}{L-BFGS-B}}
      \begin{itemize}
        \item Usa el \textbf{gradiente exacto} (diferencias finitas)
        \item Estima la \textbf{curvatura} del terreno (Hessiana aproximada)
        \item Un solo camino, muy pulido
        \item Pasos más inteligentes (tamaño adaptativo)
      \end{itemize}
      \vspace{0.1cm}
      \scriptsize
      \textbf{Analogía:} Caminas con un mapa topográfico
      — sabes la pendiente y la forma del terreno.
    \end{block}

    \column{0.5\textwidth}
    \begin{block}{\textcolor{Plum}{SGD}}
      \begin{itemize}
        \item Usa un \textbf{gradiente aproximado} ruidoso
        \item Ignora la curvatura (solo dirección)
        \item Múltiples reinicios desde distintos puntos
        \item Momentum para no quedarse atascado
      \end{itemize}
      \vspace{0.1cm}
      \scriptsize
      \textbf{Analogía:} Lanzas 50 personas desde distintos
      puntos de la montaña, cada una baja a ciegas
      sintiendo el piso.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile, shrink]{DE — Código: optimización global}
  \textbf{\textcolor{MidnightBlue}{Núcleo del optimizador:}} búsqueda evolutiva con \texttt{differential\_evolution}
  seguida de refinamiento local \texttt{L-BFGS-B}.

  \begin{pycode}[basicstyle=\ttfamily\scriptsize, numbers=none]
class DifferentialEvolutionOptimizer:
    def optimize(self, n_steps=800, x0=X0, y0=Y0):
        bounds = [ALPHA_BOUNDS, BETA_BOUNDS,
                  DELTA_BOUNDS, GAMMA_BOUNDS]

        # Fase 1: búsqueda global evolutiva
        res = differential_evolution(
            lambda p: cost_function(p, n_steps, x0, y0),
            bounds,
            strategy='best1bin',   # mutación diferencial
            maxiter=1500,          # generaciones
            popsize=50,            # individuos por generación
            tol=1e-8, seed=42,
        )

        # Fase 2: refinamiento local
        res2 = minimize(
            lambda p: cost_function(p, n_steps, x0, y0),
            res.x, bounds=bounds, method='L-BFGS-B')

        xopt = res2.x if res2.fun < res.fun else res.x
        self.a, self.b, self.c, self.d = xopt
        self.x_sim, self.y_sim = simulate(*xopt, n_steps, x0, y0)
        return self
  \end{pycode}

  \vspace{0.1cm}
  {\footnotesize \texttt{differential\_evolution}} explora el espacio de búsqueda
  de forma paralela con una población de 50 individuos.
  \texttt{L-BFGS-B} refina la mejor solución encontrada.
\end{frame}

\begin{frame}[shrink]{SGD — Stochastic Gradient Descent}
  \footnotesize
  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Estrategia:}
    \begin{itemize}
      \item Gradiente aproximado por diferencias finitas

    \textbf{Métricas:}
    \begin{itemize}
      \item Costo: $-25.34$
      \item CV: $\mathbf{0.041}$ (más estable)
      \item Ganancia: \$61.72/mes
      \item Tiempo: $19.9$s
    \end{itemize}

    \column{0.45\textwidth}
    \centering
    \begin{tikzpicture}[node distance=0.2cm]
      \node[draw, fill=Plum!20, rounded corners, inner sep=5pt, align=center] (grid) {Grid de puntos iniciales};

\begin{frame}[fragile, shrink]{SGD — Código: gradiente descendente}
  \textbf{\textcolor{Plum}{Núcleo del optimizador:}} gradiente aproximado por diferencias finitas
  con momentum y grid de reinicios.

  \vspace{0.2cm}
  \textbf{Gradiente por diferencias finitas:}
  \begin{pycode}[basicstyle=\ttfamily\scriptsize, numbers=none]
def _finite_grad(self, p, n_steps, x0, y0, eps=1e-4):
    base = cost_function(p, n_steps, x0, y0)
    grad = np.zeros(4)
    for i in range(4):         # perturba cada parámetro
        p_up = p.copy()
        p_up[i] += eps
        p_up = self._clip(p_up)
        grad[i] = (cost_function(p_up, n_steps, x0, y0) - base) / eps
    # clipping para evitar explosiones cerca de penalizaciones
    norm = np.linalg.norm(grad)
    if norm > 1000:
        grad = grad / norm * 1000
    return grad, base
  \end{pycode}

  \textbf{Bucle de descenso con momentum:}
  \begin{pycode}[basicstyle=\ttfamily\scriptsize, numbers=none]
def _restart_from(self, p0, n_steps, x0, y0, max_iter=150, lr=0.005):
    p = p0.copy();  v = np.zeros(4)      # momentum inicial
    best_p = p.copy()
    best_c = cost_function(p, n_steps, x0, y0)
    for step in range(max_iter):
        grad, _ = self._finite_grad(p, n_steps, x0, y0)
        v = 0.85 * v + lr * grad         # momentum
        p = self._clip(p - v)            # actualización
        lr *= 0.99                       # decaimiento
        ...
  \end{pycode}
\end{frame}

\section{Optimizador 3: ANFIS}

\begin{frame}[shrink]{ANFIS — Neuro-Fuzzy Optimizer}
  \footnotesize
  \begin{columns}
    \column{0.55\textwidth}
    \textbf{Estrategia:}
    \begin{itemize}
      \item 3 funciones de membresía Gaussianas por parámetro (Low, Medium, High)
      \item Red neuronal feedforward (12 $\to$ 16 $\to$ 4) ajusta centros y anchos
      \item Inferencia difusa: reglas heurísticas basadas en el costo
      \item Entrenamiento: muestreo $\to$ evaluar $\to$ retropropagar
    \end{itemize}

    \textbf{Parámetros óptimos:}
    \[
    a = 0.883,\; b = 0.013,\; c = 0.004,\; d = 0.269
    \]
    \[
    x^{*} = 63.8,\; y^{*} = 66.9
    \]

    \textbf{Métricas:}
    \begin{itemize}
      \item Costo: $-12.11$
      \item CV: $0.338$
      \item Retención: $\mathbf{76.4\%}$ (mejor)
      \item Tiempo: $\mathbf{0.5}$s (más rápido)
    \end{itemize}

    \column{0.45\textwidth}
    \centering
    \begin{tikzpicture}[node distance=0.2cm]

  \textbf{\textcolor{ForestGreen}{Núcleo del optimizador:}} fuzzificación de parámetros
  + red neuronal para ajuste fino.

  \vspace{0.2cm}
  \textbf{Fuzzificación con Gaussianas:}
  \begin{pycode}[basicstyle=\ttfamily\scriptsize, numbers=none]
def _fuzzify(self, params):
    a, b, c, d = params
    x = np.array([a, b, c, d])
    phi = []
    for i in range(4):                 # para cada parámetro
        for j in range(self.n_mf):     # Low, Medium, High
            val = exp(-((x[i] - mu[i,j])**2) / (2 * sigma[i,j]**2))
            phi.append(val)            # grado de membresía
    return np.array(phi)               # vector de 12 características
  \end{pycode}

  \textbf{Red neuronal feedforward (12→16→4):}
  \begin{pycode}[basicstyle=\ttfamily\scriptsize, numbers=none]
def _nn_predict(self, phi):
    h = np.tanh(self.W1.T @ phi + self.b1)      # capa oculta
    out = self.W2.T @ h + self.b2                # capa de salida
    return np.tanh(out) * 0.1                    # ajuste acotado

def _nn_train(self, phi, target, lr=0.01):
    pred = self._nn_predict(phi)
    error = pred - target                        # error de ajuste
    d_W2 = np.outer(h, error * (1 - tanh(out)**2) * 0.1)
    d_W1 = np.outer(phi, self.W2 @ d_out * (1 - h**2))
    self.W2 -= lr * d_W2                         # retropropagación
    self.W1 -= lr * d_W1
  \end{pycode}
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{enumerate}
    \item \textbf{DE es el mejor optimizador global:} encuentra el costo más bajo
      ($-27.95$) y la mayor ganancia (\$70.60/mes). Es la opción recomendada
      para producción.
    \bigskip
    \item \textbf{SGD es competitivo para re-optimización:} con grid de reinicios
      logra un costo cercano ($-25.34$) y la mejor estabilidad (CV=0.041),
      aunque es más lento.
    \bigskip
    \item \textbf{ANFIS ofrece un trade-off distinto:} maximiza retención (76.4\%)
      sacrificando ganancia. Su tiempo de cómputo es mínimo (0.5s).
      Útil para explorar estrategias alternativas.
    \bigskip
    \item \textbf{El equilibrio $x^{*}=d/c$, $y^{*}=a/b$ se mantiene} como atractor
      del sistema independientemente del optimizador.
  \end{enumerate}
\end{frame}
