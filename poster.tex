    \begin{block}{Planteamiento del Problema}
    \textbf{¿Cómo predecir la dinámica de sistemas presa-depredador?}
    \begin{itemize}
    \item Los sistemas dinámicos no lineales como Lotka-Volterra son difíciles de predecir analíticamente
    \item La identificación de sistemas permite aproximar su comportamiento mediante modelos
    \item La predicción multistep es relevante en ecología matemática y control de poblaciones
    \item Se comparan dos enfoques: redes neuronales (MLP) y sistemas de inferencia difusa (ANFIS)
    \end{itemize}
    \end{block}

    \vspace{0.6cm}

    \begin{block}{Modelo Matemático - Lotka-Volterra}
    Sistema de ecuaciones diferenciales ordinarias:
    \[
    \dot{x} = ax - bxy \quad\quad \dot{y} = cxy - dy
    \]
    \begin{center}
    \begin{tabular}{ll}
    \toprule
    \textbf{Variable/Parám.} & \textbf{Descripción} \\
    \midrule
    $x(t)$ & Población de presas \\
    $y(t)$ & Población de depredadores \\
    $a$ & Tasa de crecimiento de presas \\
    $b$ & Tasa de predación (captura) \\
    $c$ & Tasa de conversión de presas a depredadores \\
    $d$ & Tasa de mortalidad de depredadores \\
    \bottomrule
    \end{tabular}
    \end{center}
    \textbf{Discretización:} Método de Euler con $T=0.1$
    \end{block}

    \vspace{0.6cm}

    \begin{block}{Isomorfismo: Red Social / App de Citas}
    \centering
    \normalsize
    \setlength{\tabcolsep}{3pt}
    \renewcommand{\arraystretch}{1.1}

    \begin{tabularx}{\linewidth}{@{}p{0.33\linewidth} p{0.33\linewidth} p{0.33\linewidth}@{}}
    \toprule
    \textbf{Lotka-Volterra} & \textbf{App de Citas} & \textbf{Valor} \\
    \midrule
    Presas ($x$) & Usuarios activos & -- \\
    Depredadores ($y$) & Matches inactivos & -- \\
    $a$ & $\alpha$ (nuevos perfiles) & 0.5 \\
    $b$ & $\beta$ (match/salida) & 0.01 \\
    $c$ & $\gamma$ (abandono) & 0.2 \\
    $d$ & $\delta$ (eficiencia) & 0.005 \\
    \bottomrule
    \end{tabularx}

    \end{block}

    \vspace{0.6cm}

    \begin{block}{Arquitectura MLP (Red Neuronal)}
    \begin{itemize}
    \item \textbf{Arquitectura:} 2 entradas $\to$ (64, 64, 32) $\to$ 2 salidas
    \item \textbf{Activación:} ReLU en capas ocultas
    \item \textbf{Optimizador:} Adam con early stopping
    \item \textbf{Épocas:} 51 (con parada temprana)
    \end{itemize}
    \end{block}

    \vspace{0.6cm}

    \begin{block}{Arquitectura ANFIS}
    \begin{itemize}
    \item \textbf{Funciones de membresía:} 9 gaussianas por entrada ($\sigma=18$)
    \item \textbf{Rejilla uniforme:} $[10,90]$ $\to$ 81 reglas difusas
    \item \textbf{Regresión:} Ridge ($\alpha=0.30$) - solución analítica instantánea
    \end{itemize}
    \end{block}

    \begin{alertblock}{Conclusiones}
    \begin{itemize}
    \item \textbf{MLP:} Excelente rendimiento en predicción multistep (MAE$\approx$1.4), captura la dinámica del ciclo límite
    \item \textbf{ANFIS:} Entrenamiento instantáneo pero menor precisión en horizontes largos
    \item \textbf{Data augmentation:} Clave para generalización a condiciones iniciales diversas
    \end{itemize}
    \end{alertblock}
